CREATE DATABASE IF NOT EXISTS scolarite_abs
  CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
USE scolarite_abs;

DROP TABLE IF EXISTS absence;
DROP TABLE IF EXISTS etudiant;
DROP TABLE IF EXISTS classe;
DROP TABLE IF EXISTS departement;
DROP TABLE IF EXISTS matiere;

CREATE TABLE departement (
  cod_dept INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  nom_dept VARCHAR(50) NOT NULL
);

CREATE TABLE classe (
  cod_class INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  nom_class VARCHAR(50) NOT NULL,
  nbr_etud INT DEFAULT 0,
  cod_dept INT NOT NULL
);

CREATE TABLE etudiant (
  nce VARCHAR(20) NOT NULL PRIMARY KEY,
  nom VARCHAR(50) NOT NULL,
  prenom VARCHAR(50) NOT NULL,
  date_nais DATE DEFAULT NULL,
  lieu_nais VARCHAR(50) DEFAULT NULL,
  cod_class INT NOT NULL
);

CREATE TABLE matiere (
  cod_mat VARCHAR(20) NOT NULL PRIMARY KEY,
  int_mat VARCHAR(100) NOT NULL,
  description VARCHAR(255) DEFAULT NULL
);

CREATE TABLE absence (
  cod_mat VARCHAR(20) NOT NULL,
  nce VARCHAR(20) NOT NULL,
  date_a DATE NOT NULL,
  nha DECIMAL(4,1) NOT NULL,
  PRIMARY KEY (cod_mat, nce, date_a)
);

CREATE DATABASE IF NOT EXISTS scol_a7
  CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
USE scol_a7;

DROP TABLE IF EXISTS etudiant;
DROP TABLE IF EXISTS classe;
DROP TABLE IF EXISTS formation;
DROP TABLE IF EXISTS enseignant;
DROP TABLE IF EXISTS user;

CREATE TABLE classe (
  cod_class INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  nbre_etud INT DEFAULT NULL,
  nom_class VARCHAR(255) DEFAULT NULL
);

CREATE TABLE enseignant (
  id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  matiere VARCHAR(255) DEFAULT NULL,
  nom VARCHAR(255) DEFAULT NULL,
  prenom VARCHAR(255) DEFAULT NULL
);

CREATE TABLE formation (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  duree INT NOT NULL,
  nom VARCHAR(255) DEFAULT NULL
);

CREATE TABLE etudiant (
  id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  date_nais DATE DEFAULT NULL,
  nom VARCHAR(255) DEFAULT NULL,
  prenom VARCHAR(255) DEFAULT NULL,
  classe_cod_class INT DEFAULT NULL,
  formation_id BIGINT DEFAULT NULL,
  FOREIGN KEY (classe_cod_class) REFERENCES classe (cod_class),
  FOREIGN KEY (formation_id) REFERENCES formation (id)
);

CREATE TABLE user (
  id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  email VARCHAR(100) DEFAULT NULL,
  password VARCHAR(255) DEFAULT NULL,
  UNIQUE KEY uk_user_email (email)
);
